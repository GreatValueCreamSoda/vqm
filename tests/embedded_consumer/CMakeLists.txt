cmake_minimum_required(VERSION 3.25)
project(vqm_embedded_consumer LANGUAGES CXX)
if(NOT VQM_SOURCE_DIR)
  message(FATAL_ERROR "VQM_SOURCE_DIR is required")
endif()
set(CMAKE_DISABLE_FIND_PACKAGE_PNG TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_JPEG TRUE)
if(NOT DEFINED BUILD_TESTING)
  set(VQM_CHECK_TESTING_DEFAULT TRUE)
endif()
add_subdirectory("${VQM_SOURCE_DIR}" vqm)
if(VQM_BUILD_CLI OR VQM_BUILD_BENCH OR VQM_BUILD_TESTS)
  message(FATAL_ERROR "embedding should not enable standalone tools by default")
endif()
if(VQM_CHECK_TESTING_DEFAULT AND DEFINED BUILD_TESTING)
  message(FATAL_ERROR "embedding should not define the parent's BUILD_TESTING option")
endif()
add_executable(embedded "${VQM_SOURCE_DIR}/tests/package_consumer/vqm_main.cc")
target_link_libraries(embedded PRIVATE vqm::core)
