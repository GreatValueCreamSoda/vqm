cmake_minimum_required(VERSION 3.25)
project(vqm_consumer LANGUAGES CXX)
if(VQM_TEST_PREFIX)
  list(PREPEND CMAKE_PREFIX_PATH "${VQM_TEST_PREFIX}")
endif()
find_package(vqm CONFIG QUIET COMPONENTS unsupported)
if(vqm_FOUND)
  message(FATAL_ERROR "vqm reported an unsupported required component as available")
endif()
find_package(vqm CONFIG REQUIRED)
if(VQM_TEST_SDK_DIR)
  # Model a Vulkan SDK outside the compiler's default include paths. The
  # consumer must inherit this directory through vqm::core.
  set_property(TARGET Vulkan::Vulkan APPEND PROPERTY INTERFACE_INCLUDE_DIRECTORIES
               "${VQM_TEST_SDK_DIR}")
endif()
add_executable(consumer vqm_main.cc)
if(VQM_TEST_SDK_DIR)
  target_compile_definitions(consumer PRIVATE VQM_PROBE_VULKAN_SDK)
endif()
target_link_libraries(consumer PRIVATE vqm::core)
get_property(VQM_TEST_SHARED_LIBS GLOBAL PROPERTY TARGET_SUPPORTS_SHARED_LIBS)
if(VQM_TEST_SHARED_LIBS)
  add_library(consumer_shared SHARED shared.cc)
  set_target_properties(consumer_shared PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
  target_link_libraries(consumer_shared PRIVATE vqm::core)
endif()
enable_testing()
add_test(NAME consumer COMMAND consumer)
